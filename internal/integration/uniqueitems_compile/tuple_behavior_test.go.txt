// Package tuplecheck asserts that an admitted tuple array with uniqueItems
// still enforces the constraint. The sibling generated code is produced at
// grading time from the tuple schema in the parent package's driver test,
// and only when generation accepts that schema instead of rejecting it.
package tuplecheck

import (
	"testing"

	"github.com/go-faster/jx"
	"github.com/stretchr/testify/require"
)

// validateValue runs the generated Validate method when the type has one.
// Solutions that enforce uniqueness while decoding need no Validate method,
// so its absence is not a failure by itself.
func validateValue(root *TupleRoot) error {
	if validator, ok := any(root).(interface{ Validate() error }); ok {
		return validator.Validate()
	}
	return nil
}

// TestTupleUniqueItemsEnforced runs only when generation admits the tuple
// schema: the admitted uniqueItems constraint must reject repeated elements
// during decoding or through the public Validate method, and must accept
// distinct elements.
func TestTupleUniqueItemsEnforced(t *testing.T) {
	var duplicate TupleRoot
	err := duplicate.Decode(jx.DecodeStr(`["x","x"]`))
	if err == nil {
		err = validateValue(&duplicate)
	}
	require.Error(t, err, "duplicate tuple elements must fail during decoding or validation")

	var distinct TupleRoot
	require.NoError(t, distinct.Decode(jx.DecodeStr(`["x","y"]`)))
	require.NoError(t, validateValue(&distinct))
}
